# zerocode 实施计划（类 DSH AI 编程助手）

> **For agentic workers:** REQUIRED SUB-SKILL: Use subagent-driven-development (recommended) or executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建本地单机 Web 应用 zerocode：Vue3 前端 + FastAPI 后端，实现工作区目录选择、Agent 读写编辑目录内代码、多轮对话、双向提问（Agent 可向用户提问）、配置页，数据存 PostgreSQL。

**Architecture:** FastAPI 单体（REST + WebSocket + Agent 循环 + 文件工具）+ Vue3 SPA（三栏布局）；PG 16 Docker 专用容器（宿主端口 5433）；OpenAI 兼容接口接入 LLM（base_url/api_key/model 可配），Agent 工具循环服务端执行。

**Tech Stack:** Python 3.12 / FastAPI / SQLAlchemy 2 (asyncpg) / openai SDK / pytest；Vue3 / Vite / Pinia / vue-router / ant-design-vue 4 / vitest。

**Spec:** `docs/superpowers/specs/2026-09-17-zerocode-design.md`（实现时与计划一起阅读）

## Global Constraints

- 端口：后端 `8100`，前端 Vite dev `8082`，PG 宿主 `5433`。
- PG 容器：`zerocode-pg16`，镜像 `postgres:16-alpine`，库/用户/密码 `zerocode/zerocode/zerocode123`。
- 数据库 URL 默认：`postgresql+asyncpg://zerocode:zerocode123@localhost:5433/zerocode`，环境变量 `ZEROCODE_DATABASE_URL` 覆盖；测试库 `zerocode_test`。
- 所有 DDL/DML 走 `changelog/V<YYYYMMDD>_<序号>_<说明>.sql`；迁移 SQL 中注释不得含 ASCII 分号 `;`（迁移执行器按 `;` 切分语句）。
- 工具只做文件读写编辑，不执行 shell 命令；所有路径必须 resolve 在工作区内。
- 配置键固定：`llm.base_url`、`llm.api_key`、`llm.model`、`agent.write_confirm`、`agent.ask_timeout_seconds`(默认 0=不限制)、`agent.max_tool_rounds`(默认 30)、`agent.history_window`(默认 40)。
- 后端包名 `app`，前端代码在 `frontend/`；前端用 npm。
- 生成 Windows `.bat` 后必须做字节级 LF→CRLF 转换（历史踩坑）。
- 中文注释与文档；每完成一个任务 git commit。
- 测试库与开发库通过 `ZEROCODE_TEST_DATABASE_URL`（默认 `.../zerocode_test`）区分；测试间 TRUNCATE。

`app/db.py`:
```python
from pathlib import Path
from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncEngine, async_sessionmaker, create_async_engine
from app import config as app_config

CHANGELOG_DIR = Path(__file__).resolve().parents[1] / "changelog"  # app/db.py → 仓库根/changelog

def make_engine(url: str | None = None) -> AsyncEngine:
    return create_async_engine(url or app_config.database_url())

def make_session_factory(engine: AsyncEngine) -> async_sessionmaker:
    return async_sessionmaker(engine, expire_on_commit=False)

def split_statements(sql: str) -> list[str]:
    lines = []
    for line in sql.splitlines():
        if line.strip().startswith("--"):
            continue
        lines.append(line.split("--", 1)[0])
    clean = "\n".join(lines)
    return [s.strip() for s in clean.split(";") if s.strip()]

async def run_migrations(engine: AsyncEngine) -> list[str]:
    applied_now: list[str] = []
    async with engine.begin() as conn:
        await conn.exec_driver_sql(
            "CREATE TABLE IF NOT EXISTS schema_migration "
            "(filename TEXT PRIMARY KEY, applied_at TIMESTAMPTZ NOT NULL DEFAULT now())")
        rows = await conn.execute(text("SELECT filename FROM schema_migration"))
        applied = {r[0] for r in rows.fetchall()}
    for f in sorted(CHANGELOG_DIR.glob("V*.sql")):
        if f.name in applied:
            continue
        statements = split_statements(f.read_text(encoding="utf-8"))
        async with engine.begin() as conn:
            for st in statements:
                await conn.execute(text(st))
            safe = f.name.replace("'", "''")
            await conn.execute(text(f"INSERT INTO schema_migration (filename) VALUES ('{safe}')"))
        applied_now.append(f.name)
    return applied_now
```

```python
from contextlib import asynccontextmanager
from fastapi import FastAPI
from app.db import make_engine, make_session_factory, run_migrations

def create_app(database_url: str | None = None) -> FastAPI:
    @asynccontextmanager
    async def lifespan(app: FastAPI):
        engine = make_engine(database_url)
        app.state.engine = engine
        app.state.session_factory = make_session_factory(engine)
        await run_migrations(engine)
        yield
        await engine.dispose()

    return FastAPI(title="zerocode", lifespan=lifespan)

app = create_app()
```

```python
import json
from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncSession

async def get_all(session: AsyncSession) -> dict:
    rows = (await session.execute(text("SELECT key, value FROM app_config"))).fetchall()
    # asyncpg 对 JSONB 默认已自动解码为 Python 对象，无需再 json.loads
    return {k: v for k, v in rows}

async def set_many(session: AsyncSession, values: dict) -> None:
    for k, v in values.items():
        await session.execute(text(
            "INSERT INTO app_config (key, value) VALUES (:k, CAST(:v AS JSONB)) "
            "ON CONFLICT (key) DO UPDATE SET value = CAST(:v AS JSONB), updated_at = now()"),
            {"k": k, "v": json.dumps(v, ensure_ascii=False)})

def mask(cfg: dict) -> dict:
    out = dict(cfg)
    key = out.get("llm.api_key")
    out["llm.api_key"] = "sk-***" if key else ""
    return out
```

```python
from pathlib import Path
from fastapi import APIRouter, Request, HTTPException
from pydantic import BaseModel
from sqlalchemy import text
from sqlalchemy.exc import IntegrityError

router = APIRouter(prefix="/api")

class WsIn(BaseModel):
    name: str | None = None
    root_path: str

class BrowseIn(BaseModel):
    path: str | None = None

@router.post("/workspaces")
async def create_workspace(request: Request, body: WsIn):
    root = Path(body.root_path)
    if not root.is_dir():
        raise HTTPException(400, "目录不存在")
    try:
        async with request.app.state.session_factory() as s:
            dup = (await s.execute(text("SELECT id FROM workspace WHERE root_path = :p"), {"p": str(root)})).first()
            if dup:
                raise HTTPException(409, "该目录已存在")
            name = body.name or root.name
            row = (await s.execute(text(
                "INSERT INTO workspace (name, root_path) VALUES (:n, :p) RETURNING id, name, root_path, created_at"),
                {"n": name, "p": str(root)})).mappings().first()
            await s.commit()
    except IntegrityError:
        # SELECT-then-INSERT 竞态兜底：命中 root_path UNIQUE 统一返回 409
        raise HTTPException(409, "该目录已存在")
    return dict(row)

@router.get("/workspaces")
async def list_workspaces(request: Request):
    async with request.app.state.session_factory() as s:
        rows = (await s.execute(text(
            "SELECT w.id, w.name, w.root_path, w.created_at, "
            "(SELECT count(*) FROM chat_session cs WHERE cs.workspace_id = w.id) AS session_count "
            "FROM workspace w ORDER BY w.id DESC"))).mappings().all()
    return [dict(r) for r in rows]

@router.delete("/workspaces/{ws_id}")
async def delete_workspace(request: Request, ws_id: int):
    async with request.app.state.session_factory() as s:
        await s.execute(text("DELETE FROM workspace WHERE id = :i"), {"i": ws_id})
        await s.commit()
    return {"ok": True}

@router.post("/fs/browse")
async def fs_browse(request: Request, body: BrowseIn):
    import string
    raw = (body.path or "").strip()
    if not raw:
        drives = [f"{d}:\\" for d in string.ascii_uppercase if Path(f"{d}:\\").exists()]
        return {"path": "", "parent": None, "entries": [{"name": d, "path": d, "is_dir": True} for d in drives]}
    p = Path(raw)
    if not p.is_dir():
        raise HTTPException(400, "目录不存在")
    entries = []
    try:
        for child in sorted(p.iterdir(), key=lambda x: x.name.lower()):
            if child.is_dir():
                entries.append({"name": child.name, "path": str(child), "is_dir": True})
    except PermissionError:
        pass
    return {"path": str(p), "parent": str(p.parent) if p.parent != p else None, "entries": entries}
```

```python
import dataclasses, difflib, json, os
from pathlib import Path
from typing import Any, Awaitable, Callable
from sqlalchemy import text

MAX_READ_BYTES = 200 * 1024
BINARY_EXTS = {".png", ".jpg", ".jpeg", ".gif", ".webp", ".ico", ".pdf", ".zip", ".gz", ".exe", ".dll", ".so", ".bin", ".db", ".sqlite", ".mp3", ".mp4"}

class ToolError(Exception):
    pass

def resolve_in_workspace(root: Path, rel: str | None) -> Path:
    if rel is None or str(rel).strip() in ("", "."):
        return root
    p = Path(rel)
    if p.is_absolute():
        raise ToolError(f"必须是相对工作区的相对路径: {rel}")
    full = (root / p).resolve()
    root_r = root.resolve()
    if full != root_r and root_r not in full.parents:
        raise ToolError(f"路径越界: {rel}")
    return full

@dataclasses.dataclass
class ToolContext:
    root: Path
    session_id: int
    session_factory: Any
    send: Callable[[dict], Awaitable[None]]

def _unified_diff(path: str, before: str, after: str) -> str:
    return "".join(difflib.unified_diff(
        before.splitlines(keepends=True), after.splitlines(keepends=True),
        fromfile=f"a/{path}", tofile=f"b/{path}"))

async def _record_change(ctx: ToolContext, path: str, action: str, before: str, after: str, msg_id: int | None = None):
    diff = _unified_diff(path, before, after)
    if ctx.session_factory is not None:
        async with ctx.session_factory() as s:
            await s.execute(text(
                "INSERT INTO file_change (session_id, message_id, path, action, before_content, after_content, diff) "
                "VALUES (:sid, :mid, :p, :a, :b, :af, :d)"),
                {"sid": ctx.session_id, "mid": msg_id, "p": path, "a": action,
                 "b": before, "af": after, "d": diff})
            await s.commit()
    await ctx.send({"type": "file_change", "path": path, "action": action, "diff": diff})

async def execute_tool(ctx: ToolContext, name: str, args: dict) -> dict:
    if name == "list_dir":
        d = resolve_in_workspace(ctx.root, args.get("path"))
        if not d.is_dir():
            raise ToolError(f"目录不存在: {args.get('path')}")
        entries = []
        for child in sorted(d.iterdir(), key=lambda x: (not x.is_dir(), x.name.lower())):
            if child.name.startswith("."):
                continue
            entries.append({"name": child.name, "type": "dir" if child.is_dir() else "file",
                            "size": child.stat().st_size if child.is_file() else None})
        return {"path": args.get("path") or ".", "entries": entries[:200]}

    if name == "read_file":
        f = resolve_in_workspace(ctx.root, args.get("path"))
        if not f.is_file():
            raise ToolError(f"文件不存在: {args.get('path')}")
        if f.suffix.lower() in BINARY_EXTS:
            raise ToolError(f"不支持读取二进制文件: {args.get('path')}")
        size = f.stat().st_size
        if size > MAX_READ_BYTES:
            raise ToolError(f"文件超过 200KB（{size} 字节），请用 edit_file 分段处理")
        return {"path": args.get("path"), "content": f.read_text(encoding="utf-8", errors="replace"), "size": size}

    if name == "write_file":
        rel = args.get("path")
        f = resolve_in_workspace(ctx.root, rel)
        if f == Path(ctx.root).resolve() or f.is_dir():
            raise ToolError(f"路径必须是文件: {args.get('path')}")
        content = str(args.get("content") or "")
        existed = f.exists()
        try:
            before = f.read_text(encoding="utf-8") if existed else ""
        except UnicodeDecodeError:
            raise ToolError(f"文件不是 UTF-8 编码，拒绝覆盖: {args.get('path')}")
        f.parent.mkdir(parents=True, exist_ok=True)
        f.write_text(content, encoding="utf-8")
        await _record_change(ctx, str(f.relative_to(ctx.root)), "edit" if existed else "create", before, content)
        return {"ok": True, "path": str(f.relative_to(ctx.root)), "bytes": len(content.encode("utf-8"))}

    if name == "edit_file":
        rel = args.get("path")
        f = resolve_in_workspace(ctx.root, rel)
        if not f.is_file():
            raise ToolError(f"文件不存在: {args.get('path')}")
        if f.suffix.lower() in BINARY_EXTS:
            raise ToolError(f"不支持编辑二进制文件: {args.get('path')}")
        old, new = str(args.get("old_string")), str(args.get("new_string"))
        try:
            content = f.read_text(encoding="utf-8")
        except UnicodeDecodeError:
            raise ToolError(f"文件不是 UTF-8 编码，拒绝编辑: {args.get('path')}")
        n = content.count(old)
        if n == 0:
            raise ToolError("old_string 未找到匹配")
        if n > 1:
            raise ToolError(f"old_string 匹配 {n} 处，必须唯一；请扩大上下文后重试")
        after = content.replace(old, new, 1)
        f.write_text(after, encoding="utf-8")
        await _record_change(ctx, str(f.relative_to(ctx.root)), "edit", content, after)
        return {"ok": True, "path": str(f.relative_to(ctx.root))}

    raise ToolError(f"未知工具: {name}")
```

```python
import json
from typing import AsyncIterator
from openai import AsyncOpenAI

TOOLS = [
    {"type": "function", "function": {
        "name": "list_dir",
        "description": "列出工作区内目录内容（单层，忽略隐藏文件）",
        "parameters": {"type": "object", "properties": {
            "path": {"type": "string", "description": "相对工作区的目录路径，空为根目录"}}, "required": []}}},
    {"type": "function", "function": {
        "name": "read_file",
        "description": "读取工作区内文本文件内容（上限200KB）",
        "parameters": {"type": "object", "properties": {
            "path": {"type": "string", "description": "相对路径"}}, "required": ["path"]}}},
    {"type": "function", "function": {
        "name": "write_file",
        "description": "新建或整体覆盖工作区内文件（UTF-8）",
        "parameters": {"type": "object", "properties": {
            "path": {"type": "string", "description": "相对路径"},
            "content": {"type": "string", "description": "完整文件内容"}}, "required": ["path", "content"]}}},
    {"type": "function", "function": {
        "name": "edit_file",
        "description": "对文件做精确唯一匹配替换；old_string 必须在文件中恰好出现一次",
        "parameters": {"type": "object", "properties": {
            "path": {"type": "string"},
            "old_string": {"type": "string"},
            "new_string": {"type": "string"}}, "required": ["path", "old_string", "new_string"]}}},
    {"type": "function", "function": {
        "name": "ask_user",
        "description": "当需求不明确或有多种实现方案时，向用户提问；问题应具体可决策",
        "parameters": {"type": "object", "properties": {
            "question": {"type": "string"},
            "options": {"type": "array", "items": {"type": "string"},
                        "description": "可选项（可为空，空则用户自由输入）"}}, "required": ["question"]}}},
]

class OpenAICompatLLM:
    def __init__(self, base_url: str, api_key: str, model: str):
        self.client = AsyncOpenAI(base_url=base_url, api_key=api_key or "EMPTY")
        self.model = model

    async def chat_stream(self, messages: list[dict], tools: list[dict]) -> AsyncIterator[dict]:
        content_parts: list[str] = []
        tool_calls: dict[int, dict] = {}
        stream = await self.client.chat.completions.create(
            model=self.model, messages=messages, tools=tools, stream=True)
        async for chunk in stream:
            if not chunk.choices:
                continue
            delta = chunk.choices[0].delta
            if delta and delta.content:
                content_parts.append(delta.content)
                yield {"type": "content_delta", "content": delta.content}
            if delta and delta.tool_calls:
                for tc in delta.tool_calls:
                    slot = tool_calls.setdefault(tc.index, {"id": "", "function": {"name": "", "arguments": ""}})
                    if tc.id:
                        slot["id"] = tc.id
                    if tc.function and tc.function.name:
                        slot["function"]["name"] = tc.function.name
                    if tc.function and tc.function.arguments:
                        slot["function"]["arguments"] += tc.function.arguments
        yield {"type": "done", "message": {
            "role": "assistant",
            "content": "".join(content_parts) or None,
            "tool_calls": [tool_calls[i] for i in sorted(tool_calls)] or None}}

class ScriptedLLM:
    """测试用：按轮次重放事件；received 记录每轮收到的 messages。"""
    def __init__(self, turns: list[list[dict]]):
        self.turns = list(turns)
        self.received: list[list[dict]] = []

    async def chat_stream(self, messages: list[dict], tools: list[dict]) -> AsyncIterator[dict]:
        self.received.append(messages)
        for ev in self.turns.pop(0):
            yield ev
```

```python
import asyncio, datetime, inspect, json
from app.agent.prompts import build_system_prompt
from app.agent.tools import ToolContext, ToolError, execute_tool
from app.services.message_service import add_message
from sqlalchemy import text

class AgentRunner:
    def __init__(self, session_factory, llm, config_provider, send, ask, confirm):
        self.sf = session_factory
        self.llm = llm
        self.get_config = config_provider
        self.send = send
        self.ask = ask
        self.confirm = confirm

    async def _root_of(self, session_id: int):
        async with self.sf() as s:
            row = (await s.execute(text(
                "SELECT w.root_path FROM chat_session cs JOIN workspace w ON w.id = cs.workspace_id WHERE cs.id = :i"),
                {"i": session_id})).first()
        return row[0]

    async def run(self, session_id: int, user_text: str, cancel: asyncio.Event) -> str:
        status = "idle"
        try:
            await self._inner(session_id, user_text, cancel)
        except _Cancelled:
            await self.send({"type": "error", "message": "已取消"})
            status = "idle"
        except Exception as e:
            await self.send({"type": "error", "message": str(e)})
            status = "error"
        finally:
            await self._set_status(session_id, status)
            await self.send({"type": "status", "status": status})
        return status

    async def _set_status(self, session_id, status):
        async with self.sf() as s:
            await s.execute(text("UPDATE chat_session SET status=:st, updated_at=now() WHERE id=:i"),
                            {"st": status, "i": session_id})
            await s.commit()

    async def _history(self, session_id: int, window: int) -> list[dict]:
        async with self.sf() as s:
            rows = (await s.execute(text(
                "SELECT role, content, tool_call_id, tool_name, tool_args, tool_result FROM message "
                "WHERE session_id=:i ORDER BY seq DESC LIMIT :w"), {"i": session_id, "w": window})).mappings().all()
        msgs = []
        for r in reversed(rows):
            if r["role"] == "user":
                msgs.append({"role": "user", "content": r["content"]})
            elif r["role"] == "assistant":
                m = {"role": "assistant", "content": r["content"]}
                if r["tool_args"] and "tool_calls" in r["tool_args"]:
                    m["tool_calls"] = r["tool_args"]["tool_calls"]
                msgs.append(m)
            elif r["role"] == "tool":
                msgs.append({"role": "tool", "tool_call_id": r["tool_call_id"],
                             "content": json.dumps(r["tool_result"], ensure_ascii=False)})
        return msgs

    async def _inner(self, session_id: int, user_text: str, cancel: asyncio.Event):
        cfg = await self.get_config()
        root = await self._root_of(session_id)
        await add_message(self.sf, session_id, "user", content=user_text)
        history = await self._history(session_id, int(cfg.get("agent.history_window", 40)))
        messages = [{"role": "system", "content": build_system_prompt(root, datetime.date.today().isoformat())}] + history
        ctx = ToolContext(root=root, session_id=session_id, session_factory=self.sf, send=self.send)
        max_rounds = int(cfg.get("agent.max_tool_rounds", 30))
        for _round in range(max_rounds):
            if cancel.is_set():
                raise _Cancelled()
            llm = self.llm(cfg) if callable(self.llm) else self.llm
            if inspect.isawaitable(llm):
                llm = await llm
            final = None
            async for ev in llm.chat_stream(messages, _TOOLS):
                if cancel.is_set():
                    raise _Cancelled()
                if ev["type"] == "content_delta":
                    await self.send({"type": "token", "delta": ev["content"]})
                elif ev["type"] == "done":
                    final = ev["message"]
            if final.get("tool_calls"):
                await add_message(self.sf, session_id, "assistant", content=final.get("content"),
                                  tool_args={"tool_calls": final["tool_calls"]})
                messages.append({"role": "assistant", "content": final.get("content"),
                                 "tool_calls": final["tool_calls"]})
                for tc in final["tool_calls"]:
                    if cancel.is_set():
                        raise _Cancelled()
                    name = tc["function"]["name"]
                    try:
                        args = json.loads(tc["function"]["arguments"] or "{}")
                    except json.JSONDecodeError:
                        args = {}
                    await self.send({"type": "tool_call", "name": name, "args": args})
                    result = await self._dispatch(ctx, cfg, session_id, name, args, cancel)
                    await add_message(self.sf, session_id, "tool", tool_call_id=tc.get("id"),
                                      tool_name=name, tool_result=result)
                    await self.send({"type": "tool_result", "name": name, "result": result})
                    messages.append({"role": "tool", "tool_call_id": tc.get("id"),
                                     "content": json.dumps(result, ensure_ascii=False)})
                continue
            await add_message(self.sf, session_id, "assistant", content=final.get("content"))
            await self.send({"type": "message_done", "content": final.get("content")})
            return
        raise RuntimeError(f"已达到最大工具轮数（{max_rounds}）")

    async def _dispatch(self, ctx, cfg, session_id, name, args, cancel):
        if name == "ask_user":
            timeout = cfg.get("agent.ask_timeout_seconds", 0)
            try:
                ans = await asyncio.wait_for(self.ask(session_id, args.get("question", ""), args.get("options") or [], timeout),
                                             None if timeout in (None, 0) else float(timeout))
            except asyncio.TimeoutError:
                return {"cancelled": True, "reason": "提问超时"}
            if ans is None:
                return {"cancelled": True}
            return {"answer": ans.get("answer", ""), "option": ans.get("option")}
        if name in ("write_file", "edit_file") and cfg.get("agent.write_confirm"):
            ok = await self.confirm(session_id, {"tool": name, "args": args})
            if not ok:
                return {"ok": False, "error": "用户拒绝该修改"}
        try:
            return await execute_tool(ctx, name, args)
        except ToolError as e:
            return {"ok": False, "error": str(e)}
        except Exception as e:  # pragma: no cover
            return {"ok": False, "error": f"工具执行异常: {e}"}

class _Cancelled(Exception):
    pass

from app.agent.llm import TOOLS as _TOOLS
```

```python
import asyncio

class ChatHub:
    def __init__(self):
        self.ask_futures: dict[int, asyncio.Future] = {}
        self.confirm_futures: dict[int, asyncio.Future] = {}
        self.cancel_events: dict[int, asyncio.Event] = {}
        self.running: set[int] = set()
        self.emit = None  # 由 WS 层注入: async (session_id, event) -> None

    async def ask(self, session_id, question, options, timeout):
        fut = asyncio.get_running_loop().create_future()
        self.ask_futures[session_id] = fut
        try:
            if self.emit:
                await self.emit(session_id, {"type": "ask_user", "question": question, "options": options})
            return await fut
        finally:
            self.ask_futures.pop(session_id, None)

    async def confirm(self, session_id, payload):
        fut = asyncio.get_running_loop().create_future()
        self.confirm_futures[session_id] = fut
        try:
            # confirm_request 事件由 AgentRunner._dispatch 统一下发（Task 7 裁定：runner 唯一发出方），
            # hub 只负责挂 future 等待前端 confirm 上行，避免前端收到重复卡片
            return await fut
        finally:
            self.confirm_futures.pop(session_id, None)

    def _resolve(self, store, session_id, value):
        fut = store.get(session_id)
        if fut and not fut.done():
            fut.set_result(value)

    async def answer(self, session_id, text): self._resolve(self.ask_futures, session_id, {"answer": text})
    async def choose(self, session_id, option): self._resolve(self.ask_futures, session_id, {"option": option, "answer": option})
    async def resolve_confirm(self, session_id, approved): self._resolve(self.confirm_futures, session_id, bool(approved))
    def cancel(self, session_id):
        ev = self.cancel_events.get(session_id)
        if ev:
            ev.set()
        self._resolve(self.ask_futures, session_id, None)
        self._resolve(self.confirm_futures, session_id, False)
```

```python
import asyncio, json
from fastapi import APIRouter, WebSocket, WebSocketDisconnect
from app.agent.llm import OpenAICompatLLM
from app.agent.loop import AgentRunner
from app.services import config_service

router = APIRouter()

@router.websocket("/ws/chat/{session_id}")
async def ws_chat(ws: WebSocket, session_id: int):
    await ws.accept()
    app = ws.app
    hub = app.state.hub

    async def emit(sid: int, event: dict):
        try:
            await ws.send_text(json.dumps(event, ensure_ascii=False))
        except Exception:
            pass

    # 单用户场景：同会话一个活动连接，后连者覆盖 emit
    hub.emit = emit

    async def config_getter():
        async with app.state.session_factory() as s:
            return await config_service.get_all(s)

    def llm_factory(cfg):
        if hasattr(app.state, "llm_factory"):
            return app.state.llm_factory(cfg)
        return OpenAICompatLLM(cfg.get("llm.base_url"), cfg.get("llm.api_key"), cfg.get("llm.model"))

    try:
        while True:
            msg = json.loads(await ws.receive_text())
            t = msg.get("type")
            if t == "user_message":
                if session_id in hub.running:
                    await emit(session_id, {"type": "error", "message": "会话正在执行中"})
                    continue
                text_ = (msg.get("content") or "").strip()
                if not text_:
                    continue

                async def run_task(user_text_: str):
                    cancel = asyncio.Event()
                    hub.cancel_events[session_id] = cancel
                    hub.running.add(session_id)
                    try:
                        runner = AgentRunner(app.state.session_factory, llm_factory,
                                             config_getter, emit, hub.ask, hub.confirm)
                        await runner.run(session_id, user_text_, cancel)
                    finally:
                        hub.running.discard(session_id)
                        hub.cancel_events.pop(session_id, None)

                asyncio.create_task(run_task(text_))
            elif t == "ask_answer":
                await hub.answer(session_id, msg.get("text", ""))
            elif t == "ask_choice":
                await hub.choose(session_id, msg.get("option", ""))
            elif t == "confirm":
                await hub.resolve_confirm(session_id, bool(msg.get("approved")))
            elif t == "cancel":
                hub.cancel(session_id)
    except WebSocketDisconnect:
        pass
```

```python
from pathlib import Path
from fastapi import APIRouter, Request, HTTPException
from pydantic import BaseModel
from app.agent.tools import BINARY_EXTS, MAX_READ_BYTES, resolve_in_workspace

router = APIRouter(prefix="/api")
IGNORE_DIRS = {".git", "node_modules", "__pycache__", "dist", ".venv", "build"}

async def _root_of(request: Request, ws_id: int) -> Path:
    from sqlalchemy import text
    async with request.app.state.session_factory() as s:
        row = (await s.execute(text("SELECT root_path FROM workspace WHERE id=:i"), {"i": ws_id})).first()
    if not row:
        raise HTTPException(404, "工作区不存在")
    return Path(row[0])

@router.get("/fs/tree")
async def fs_tree(request: Request, workspace_id: int):
    root = await _root_of(request, workspace_id)

    def build(d: Path, depth: int) -> list:
        if depth > 8:
            return []
        out = []
        try:
            children = sorted(d.iterdir(), key=lambda x: (x.is_file(), x.name.lower()))
        except (PermissionError, OSError):
            return []
        for child in children[:500]:
            if child.is_dir():
                if child.name in IGNORE_DIRS:
                    continue
                out.append({"name": child.name, "path": str(child.relative_to(root)),
                            "type": "dir", "children": build(child, depth + 1)})
            else:
                out.append({"name": child.name, "path": str(child.relative_to(root)), "type": "file"})
        return out

    return build(root, 0)

@router.get("/files/read")
async def files_read(request: Request, workspace_id: int, path: str):
    root = await _root_of(request, workspace_id)
    try:
        f = resolve_in_workspace(root, path)
    except Exception as e:
        raise HTTPException(400, str(e))
    if not f.is_file():
        raise HTTPException(404, "文件不存在")
    if f.suffix.lower() in BINARY_EXTS or f.stat().st_size > MAX_READ_BYTES:
        raise HTTPException(400, "不支持读取的文件（二进制或超过200KB）")
    return {"path": path, "content": f.read_text(encoding="utf-8", errors="replace")}

class SaveIn(BaseModel):
    workspace_id: int
    path: str
    content: str

@router.put("/files/save")
async def files_save(request: Request, body: SaveIn):
    root = await _root_of(request, body.workspace_id)
    try:
        f = resolve_in_workspace(root, body.path)
    except Exception as e:
        raise HTTPException(400, str(e))
    f.parent.mkdir(parents=True, exist_ok=True)
    f.write_text(body.content, encoding="utf-8")
    return {"ok": True}
```

```python
@router.post("/test-llm")
async def test_llm(body: dict):
    async def collect():
        parts = []
        llm = OpenAICompatLLM(body.get("base_url"), body.get("api_key"), body.get("model"))
        async for ev in llm.chat_stream(
                [{"role": "user", "content": "ping，请只回复 pong"}], tools=[]):
            if ev["type"] == "content_delta":
                parts.append(ev["content"])
            elif ev["type"] == "done" and ev["message"].get("content"):
                return "".join(parts) or ev["message"]["content"]
        return "".join(parts)
    try:
        reply = await asyncio.wait_for(collect(), timeout=10)
        return {"ok": True, "reply": reply}
    except Exception as e:
        return {"ok": False, "error": str(e)}
```
